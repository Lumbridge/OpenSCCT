cmake_minimum_required(VERSION 3.24)
project(OpenScctAudioNative VERSION 1.0 LANGUAGES C CXX)
include(FetchContent)
find_package(Python3 REQUIRED COMPONENTS Interpreter)

# Keep the build machine's folders out of the shipped library. __FILE__ (godot-cpp's error macros) names sources by
# absolute path, so trim the repo, the build folder and any dependency source folders given on the command line.
# Set before the dependencies are added so their targets inherit it.
get_filename_component(SCCT_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SCCT_TRIM_PREFIXES "${SCCT_REPO_ROOT}" "${CMAKE_BINARY_DIR}")
foreach(dependency GODOTCPP OPENAL REAL)
    if(FETCHCONTENT_SOURCE_DIR_${dependency})
        get_filename_component(dependency_dir "${FETCHCONTENT_SOURCE_DIR_${dependency}}" ABSOLUTE)
        list(APPEND SCCT_TRIM_PREFIXES "${dependency_dir}")
    endif()
endforeach()
foreach(prefix IN LISTS SCCT_TRIM_PREFIXES)
    if(MSVC)
        file(TO_NATIVE_PATH "${prefix}" native_prefix)
        add_compile_options("/d1trimfile:${native_prefix}")
    else()
        add_compile_options("-ffile-prefix-map=${prefix}=.")
    endif()
endforeach()

# Exact upstream commits. Sources and generated binaries live under ignored local/ and game/addons/*/bin/.
FetchContent_Declare(godotcpp GIT_REPOSITORY https://github.com/godotengine/godot-cpp.git
    GIT_TAG e103d348c4b2d8df7b819126d855e19381cb9a87)
set(GODOTCPP_API_VERSION "4.7" CACHE STRING "Godot API")
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/godot-build-profile.json" CACHE FILEPATH "Minimal audio bindings")
set(GODOTCPP_TARGET template_release CACHE STRING "Godot extension target")
FetchContent_MakeAvailable(godotcpp)

set(SCCT_AUDIO_REVERB_BACKEND "HRVB" CACHE STRING "Native reverb backend: HRVB or OPENAL")
set_property(CACHE SCCT_AUDIO_REVERB_BACKEND PROPERTY STRINGS HRVB OPENAL)
if(SCCT_AUDIO_REVERB_BACKEND STREQUAL "HRVB")
include(hrvb.cmake)
elseif(SCCT_AUDIO_REVERB_BACKEND STREQUAL "OPENAL")
FetchContent_Declare(openal GIT_REPOSITORY https://github.com/kcat/openal-soft.git
    GIT_TAG dc7d7054a5b4f3bec1dc23a42fd616a0847af948)
set(LIBTYPE SHARED CACHE STRING "OpenAL remains replaceable for LGPL compliance" FORCE)
set(ALSOFT_UTILS OFF CACHE BOOL "" FORCE)
set(ALSOFT_EXAMPLES OFF CACHE BOOL "" FORCE)
set(ALSOFT_TESTS OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL_CONFIG OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL_HRTF_DATA OFF CACHE BOOL "" FORCE)
set(ALSOFT_INSTALL_AMBDEC_PRESETS OFF CACHE BOOL "" FORCE)
set(ALSOFT_EAX OFF CACHE BOOL "Legacy EAX API is not used; EFX EAXReverb remains available" FORCE)
set(ALSOFT_EMBED_HRTF_DATA OFF CACHE BOOL "" FORCE)
set(ALSOFT_UPDATE_BUILD_VERSION OFF CACHE BOOL "" FORCE)
# Only the offline loopback renderer is used; Godot owns the physical output device.
foreach(backend ALSA OSS SOLARIS SNDIO JACK PULSEAUDIO PIPEWIRE WASAPI DSOUND WINMM COREAudio OPENSL OBOE PORTAUDIO SDL2 OTHERIO)
    set(ALSOFT_BACKEND_${backend} OFF CACHE BOOL "" FORCE)
endforeach()
FetchContent_MakeAvailable(openal)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/patch_openal_callback.py"
    "${openal_SOURCE_DIR}/core/voice.cpp" COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/audio/package-native-sources.py"
    "${openal_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/patch_openal_callback.py"
    "${CMAKE_CURRENT_BINARY_DIR}/OpenAL-Soft-1.24.3-source.zip" COMMAND_ERROR_IS_FATAL ANY)
set(SCCT_DSP_SOURCE scct_reverb_dsp.cpp)
set(SCCT_BACKEND_LIBRARIES OpenAL::OpenAL)
set(SCCT_BACKEND_DESCRIPTION "OpenAL Soft https://github.com/kcat/openal-soft dc7d7054a5b4f3bec1dc23a42fd616a0847af948 (1.24.3), dynamically linked. Modified only core/voice.cpp synchronous unity-rate callback adapter; patch_openal_callback.py included. Reverb DSP algorithm unchanged. Complete corresponding source shipped as OpenAL-Soft-1.24.3-source.zip.")
else()
message(FATAL_ERROR "Unknown native reverb backend: ${SCCT_AUDIO_REVERB_BACKEND}")
endif()

add_library(scct_audio SHARED scct_audio.cpp ${SCCT_DSP_SOURCE})
target_compile_features(scct_audio PRIVATE cxx_std_20)
target_link_libraries(scct_audio PRIVATE godot::cpp ${SCCT_BACKEND_LIBRARIES})
if(SCCT_AUDIO_REVERB_BACKEND STREQUAL "HRVB")
    target_compile_definitions(scct_audio PRIVATE SCCT_REVERB_HRVB=1)
    if(MSVC)
        target_compile_options(scct_audio PRIVATE /fp:strict)
    else()
        target_compile_options(scct_audio PRIVATE -fno-fast-math -ffp-contract=off)
    endif()
endif()
target_include_directories(scct_audio PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(scct_audio PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES
    BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
if(WIN32)
    set(SCCT_AUDIO_PLATFORM windows-x86_64)
elseif(UNIX AND NOT APPLE AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
    set(SCCT_AUDIO_PLATFORM linux-x86_64)
else()
    message(FATAL_ERROR "Native audio currently supports Windows x86_64 and Linux x86_64")
endif()
install(TARGETS scct_audio RUNTIME DESTINATION "bin/${SCCT_AUDIO_PLATFORM}"
    LIBRARY DESTINATION "bin/${SCCT_AUDIO_PLATFORM}")
install(FILES "${godotcpp_SOURCE_DIR}/LICENSE.md" DESTINATION licenses RENAME godot-cpp-LICENSE.md)
if(SCCT_AUDIO_REVERB_BACKEND STREQUAL "OPENAL")
install(TARGETS OpenAL RUNTIME DESTINATION "bin/${SCCT_AUDIO_PLATFORM}" LIBRARY DESTINATION "bin/${SCCT_AUDIO_PLATFORM}")
install(FILES "${openal_SOURCE_DIR}/COPYING" DESTINATION licenses RENAME OpenAL-Soft-COPYING.txt)
install(FILES "${openal_SOURCE_DIR}/LICENSE-pffft" DESTINATION licenses)
install(FILES "${openal_SOURCE_DIR}/fmt-11.1.1/LICENSE" DESTINATION licenses RENAME fmt-LICENSE.txt)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/patch_openal_callback.py" DESTINATION licenses)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/OpenAL-Soft-1.24.3-source.zip" DESTINATION licenses)
else()
install(FILES "${real_SOURCE_DIR}/LICENSE" DESTINATION licenses RENAME reAL-LICENSE.txt)
install(FILES "${real_SOURCE_DIR}/LICENSE-HRTF" DESTINATION licenses RENAME reAL-HRTF-LICENSE.txt)
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/native-audio-sources.txt"
    "godot-cpp https://github.com/godotengine/godot-cpp e103d348c4b2d8df7b819126d855e19381cb9a87\n${SCCT_BACKEND_DESCRIPTION}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/native-audio-sources.txt" DESTINATION licenses)
